{{- /* صورة page resource متجاوبة. Hugo يولّد AVIF وWebP عند البناء ويحفظ المخرجات في cache، مع أصل الصورة كخيار احتياطي للمتصفحات الأقدم. `widths` اختياري لصورة صغيرة دائمًا (مصغّرات العمود الجانبي): أكبرها هو السقف. بدونه 480 و800 و1200 كما كانت. */ -}} {{- $image := .image -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "responsive-image" -}} {{- $loading := .loading | default "lazy" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $fetchpriority := .fetchpriority | default "auto" -}} {{- $candidates := .widths | default (slice 480 800 1200) -}} {{- $cap := index $candidates (sub (len $candidates) 1) -}} {{- /* Opt-in only for article cover/body, never linked related-card thumbs. Original resource is fetched only on click; default reading stays small. */ -}} {{- if .preview -}} {{- end -}} {{- if reflect.IsImageResourceProcessable $image -}} {{- $widths := slice -}} {{- range $candidates -}} {{- if le . $image.Width }}{{ $widths = $widths | append . }}{{ end -}} {{- end -}} {{- $maxWidth := math.Min $image.Width $cap | int -}} {{- $widths = $widths | append $maxWidth | uniq | sort -}} {{- $avifSet := slice -}} {{- $webpSet := slice -}} {{- range $widths -}} {{- $avif := $image.Resize (printf "%dx avif q65" .) -}} {{- $webp := $image.Resize (printf "%dx webp q78" .) -}} {{- $avifSet = $avifSet | append (printf "%s %dw" $avif.RelPermalink $avif.Width) -}} {{- $webpSet = $webpSet | append (printf "%s %dw" $webp.RelPermalink $webp.Width) -}} {{- end -}} {{- $fallback := $image -}} {{- if gt $image.Width $cap }}{{ $fallback = $image.Resize (printf "%dx" $cap) }}{{ end -}} {{- else -}} {{- end -}} {{- if .preview }}{{ end -}}
